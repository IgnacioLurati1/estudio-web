    Registrar un nuevo pedido

    Fabrica es la clase principal y guarda las colecciones, como la clase principal de los ejercicios resueltos. El número sale del último valor, que guarda la variable de clase. Al terminar se informa el número.

    "Fabrica, métodos de instancia"

#inicializa
productos := OrderedCollection new.
clientes := OrderedCollection new.
pedidos := OrderedCollection new.

#registrarPedido
| cli pedido prod cant seguir |
cli := self buscarOCrearCliente.
pedido := Pedido new.
pedido inicializa.
pedido cargaDatos: cli.
seguir := true.
[seguir] whileTrue: [
    prod := self buscarProducto.
    prod isNil
        ifTrue: [MessageBox notify: 'No existe un producto con ese código']
        ifFalse: [
            cant := (Prompter prompt: 'Ingrese la cantidad solicitada') asNumber.
            pedido agregar: prod cantidad: cant].
    seguir := MessageBox confirm: '¿Desea agregar otro producto?'].
pedido items isEmpty
    ifTrue: [MessageBox notify: 'El pedido no tiene productos y no se registró']
    ifFalse: [
        pedidos add: pedido.
        MessageBox notify: 'Pedido registrado con el número ', pedido numero printString]

#buscarOCrearCliente
| cuit cli |
cuit := Prompter prompt: 'Ingrese el CUIT del cliente'.
cli := clientes detect: [:unCliente | unCliente cuit = cuit] ifNone: [nil].
cli isNil ifTrue: [cli := self nuevoCliente: cuit].
^cli

#nuevoCliente: unCuit
| cli |
cli := Cliente new.
cli cargaDatos: unCuit.
clientes add: cli.
^cli

#buscarProducto
| cod |
cod := (Prompter prompt: 'Ingrese el código del producto') asNumber.
^productos detect: [:unProd | unProd codigo = cod] ifNone: [nil]

    Listado de pedidos de una fecha

    Se ordena por la dirección del cliente, que es un texto y se compara con <, y después se filtran los pedidos de la fecha ingresada.

    "Fabrica, métodos de instancia"

#listarPedidos
| dia lista |
dia := Date fromString: (Prompter prompt: 'Ingrese la fecha (dd/mm/aaaa)').
lista := (pedidos asSortedCollection: [:a :b | a cliente direccion < b cliente direccion])
    select: [:unPedido | unPedido fecha = dia].
lista isEmpty
    ifTrue: [MessageBox notify: 'No se han encontrado pedidos en esa fecha']
    ifFalse: [
        Transcript clear;
            show: 'PEDIDOS REGISTRADOS EL ', dia printString; cr;
            show: 'Nro.'; tab; show: 'Cliente'; tab; show: 'Dirección'; tab; show: 'Total'; cr.
        lista do: [:unPedido | unPedido mostrarEnListado]]

^tot

    show: cliente nombre; tab;
    show: cliente direccion; tab;
    show: self total printString; cr
