    El ejemplo es un sistema de préstamos de una biblioteca. La regla de negocio es que un socio no puede tener más de 3 préstamos sin devolver, y que un libro prestado no se presta de nuevo hasta que vuelva.

    Modelo del dominio

    ClaseAtributosRelación
    Socioid, nombreUn socio tiene 0 a N préstamos
    Libroid, tituloUn libro está en 0 a N préstamos, como mucho uno sin devolver
    Préstamoid, socio_id, libro_id, devueltoCada préstamo es de 1 socio y 1 libro

    Capa de datos, datos.py

    Acceso directo a SQLite con sqlite3. Los valores van con ?, para evitar la inyección SQL, y cada cambio se confirma con commit().

    import sqlite3


class RepositorioPrestamos:
    """Capa de datos: guarda y consulta socios, libros y préstamos."""

    def __init__(self, archivo='biblioteca.db'):
        # check_same_thread=False deja usar la conexión desde los hilos del servidor de Flask
        self.con = sqlite3.connect(archivo, check_same_thread=False)
        self.con.execute('CREATE TABLE IF NOT EXISTS socio (id INTEGER PRIMARY KEY, nombre TEXT NOT NULL)')
        self.con.execute('CREATE TABLE IF NOT EXISTS libro (id INTEGER PRIMARY KEY, titulo TEXT NOT NULL)')
        self.con.execute('CREATE TABLE IF NOT EXISTS prestamo (id INTEGER PRIMARY KEY, '
                         'socio_id INTEGER NOT NULL REFERENCES socio(id), '
                         'libro_id INTEGER NOT NULL REFERENCES libro(id), '
                         'devuelto INTEGER NOT NULL DEFAULT 0)')
        self.con.commit()

    def agregar_socio(self, nombre):
        cur = self.con.execute('INSERT INTO socio (nombre) VALUES (?)', (nombre,))
        self.con.commit()
        return cur.lastrowid

    def agregar_libro(self, titulo):
        cur = self.con.execute('INSERT INTO libro (titulo) VALUES (?)', (titulo,))
        self.con.commit()
        return cur.lastrowid

    def listar_libros(self):
        cur = self.con.execute('SELECT id, titulo FROM libro ORDER BY titulo')
        return cur.fetchall()

    def contar_prestamos_activos(self, socio_id):
        cur = self.con.execute('SELECT COUNT(*) FROM prestamo WHERE socio_id = ? AND devuelto = 0', (socio_id,))
        return cur.fetchone()[0]

    def libro_prestado(self, libro_id):
        cur = self.con.execute('SELECT COUNT(*) FROM prestamo WHERE libro_id = ? AND devuelto = 0', (libro_id,))
        return cur.fetchone()[0] > 0

    def registrar_prestamo(self, socio_id, libro_id):
        cur = self.con.execute('INSERT INTO prestamo (socio_id, libro_id) VALUES (?, ?)', (socio_id, libro_id))
        self.con.commit()
        return cur.lastrowid

    def devolver(self, prestamo_id):
        self.con.execute('UPDATE prestamo SET devuelto = 1 WHERE id = ?', (prestamo_id,))
        self.con.commit()

    Capa de negocio, negocio.py

    La clase de negocio evalúa las reglas antes de pedirle a la capa de datos que guarde. Si una no se cumple, lanza una excepción propia, una clase que hereda de Exception.

    class ReglaNegocioError(Exception):
    pass


class ServicioPrestamos:
    """Capa de negocio: decide si un préstamo se puede hacer."""
    MAX_PRESTAMOS = 3

    def __init__(self, repo):
        self.repo = repo

    def prestar(self, socio_id, libro_id):
        if self.repo.libro_prestado(libro_id):
            raise ReglaNegocioError('El libro ya está prestado')
        if self.repo.contar_prestamos_activos(socio_id) >= self.MAX_PRESTAMOS:
            raise ReglaNegocioError('El socio ya tiene 3 préstamos sin devolver')
        return self.repo.registrar_prestamo(socio_id, libro_id)

    El test de la regla, test_negocio.py

    Cada test arma una base en memoria con ':memory:', así no toca la base real. Se corre con python -m unittest test_negocio.

    import unittest
from datos import RepositorioPrestamos
from negocio import ServicioPrestamos, ReglaNegocioError


class TestPrestamos(unittest.TestCase):
    def setUp(self):
        self.repo = RepositorioPrestamos(':memory:')
        self.servicio = ServicioPrestamos(self.repo)
        self.socio = self.repo.agregar_socio('Socio de prueba')
        self.libros = [self.repo.agregar_libro('Libro %d' % i) for i in range(5)]

    def test_presta_hasta_tres(self):
        for libro in self.libros[:3]:
            self.servicio.prestar(self.socio, libro)
        self.assertEqual(self.repo.contar_prestamos_activos(self.socio), 3)

    def test_no_presta_el_cuarto(self):
        for libro in self.libros[:3]:
            self.servicio.prestar(self.socio, libro)
        with self.assertRaises(ReglaNegocioError):
            self.servicio.prestar(self.socio, self.libros[3])

    def test_devolver_libera_un_lugar(self):
        ids = [self.servicio.prestar(self.socio, libro) for libro in self.libros[:3]]
        self.repo.devolver(ids[0])
        self.servicio.prestar(self.socio, self.libros[3])
        self.assertEqual(self.repo.contar_prestamos_activos(self.socio), 3)

    def test_no_presta_un_libro_prestado(self):
        otro = self.repo.agregar_socio('Otro socio')
        self.servicio.prestar(self.socio, self.libros[0])
        with self.assertRaises(ReglaNegocioError):
            self.servicio.prestar(otro, self.libros[0])


if __name__ == '__main__':
    unittest.main()

    Interfaz web, app.py y templates/libros.html

    La interfaz solo pide y muestra. No decide nada, eso lo hace la capa de negocio.

    from flask import Flask, request, render_template
from datos import RepositorioPrestamos
from negocio import ServicioPrestamos, ReglaNegocioError

app = Flask(__name__)
repo = RepositorioPrestamos('biblioteca.db')
servicio = ServicioPrestamos(repo)


@app.route('/')
def inicio():
    return render_template('libros.html', libros=repo.listar_libros(), mensaje='')


@app.route('/prestar', methods=['POST'])
def prestar():
    try:
        servicio.prestar(int(request.form['socio']), int(request.form['libro']))
        mensaje = 'Préstamo registrado'
    except ValueError:
        mensaje = 'El socio y el libro van con su número'
    except ReglaNegocioError as error:
        mensaje = str(error)
    return render_template('libros.html', libros=repo.listar_libros(), mensaje=mensaje)


if __name__ == '__main__':
    app.run(debug=True)
