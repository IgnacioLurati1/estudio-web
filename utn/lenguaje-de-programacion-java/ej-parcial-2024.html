    Evento es abstracta. getPrecioFinal() es igual para todos los eventos y deja abstracto getPrecioEvento(), que cambia según cómo se factura. El catering depende de los invitados y de la duración, así que getPrecio recibe el evento.

    import java.time.LocalDate;
import java.util.ArrayList;

public abstract class Evento {

    private String nombre;
    private LocalDate fecha;
    private double duracionHs;
    private Locacion locacion;
    private ServicioCatering servicioCatering;
    private ArrayList<Invitado> invitados = new ArrayList<Invitado>();

    public Evento(String nombre, LocalDate fecha, double duracionHs, Locacion locacion, ServicioCatering servicioCatering) {
        this.nombre = nombre;
        this.fecha = fecha;
        this.duracionHs = duracionHs;
        this.locacion = locacion;
        this.servicioCatering = servicioCatering;
    }

    public void agregarInvitado(Invitado invitado) {
        invitados.add(invitado);
    }

    public int getCantidadInvitados() {
        return invitados.size();
    }

    public double getDuracionHs() {
        return duracionHs;
    }

    public abstract double getPrecioEvento();

    public double getPrecioFinal() {
        return locacion.getPrecio() + getPrecioEvento() + servicioCatering.getPrecio(this);
    }
}

    El facturado por invitado multiplica la cantidad de invitados por el precio, y si son menos que el mínimo usa el mínimo.

    import java.time.LocalDate;

public class EventoFacturadoPorInvitado extends Evento {

    private double precioPorInvitado;
    private int invitadosMinimo;

    public EventoFacturadoPorInvitado(String nombre, LocalDate fecha, double duracionHs, Locacion locacion,
            ServicioCatering servicioCatering, double precioPorInvitado, int invitadosMinimo) {
        super(nombre, fecha, duracionHs, locacion, servicioCatering);
        this.precioPorInvitado = precioPorInvitado;
        this.invitadosMinimo = invitadosMinimo;
    }

    @Override
    public double getPrecioEvento() {
        int cantidad = getCantidadInvitados();
        if (cantidad < invitadosMinimo) {
            cantidad = invitadosMinimo;
        }
        return cantidad * precioPorInvitado;
    }
}

    El catering normal sigue la fórmula de la hoja.

    precioMenuIndividual⋅invitados+∑meserosprecio por hora⋅duracionHs\text{precioMenuIndividual} \cdot \text{invitados} + \sum_{\text{meseros}} \text{precio por hora} \cdot \text{duracionHs}precioMenuIndividual⋅invitados+meseros∑​precio por hora⋅duracionHs

    import java.util.ArrayList;

public class ServicioCatering {

    private String nombre;
    private String menu;
    private double precioMenuIndividual;
    private ArrayList<Mesero> meseros = new ArrayList<Mesero>();

    public ServicioCatering(String nombre, String menu, double precioMenuIndividual) {
        this.nombre = nombre;
        this.menu = menu;
        this.precioMenuIndividual = precioMenuIndividual;
    }

    public void agregarMesero(Mesero mesero) {
        meseros.add(mesero);
    }

    public double getPrecio(Evento evento) {
        double precioMeseros = 0;
        for (Mesero m : meseros) {
            precioMeseros += m.getPrecioPorHora() * evento.getDuracionHs();
        }
        return precioMenuIndividual * evento.getCantidadInvitados() + precioMeseros;
    }
}

    Ejercicio 2

    Los nombres de los métodos son los del diagrama, getDuracion() y getTerminada(), aunque el atributo se llame duracionHs. El rombo negro de la relación se programa como la LinkedList de tareas que trae Tecnico.

    import java.util.LinkedList;

public class Tecnico {

    private String nombre;
    private String apellido;
    private LinkedList<Tarea> tareas = new LinkedList<Tarea>();

    public Tecnico(String nombre, String apellido) {
        this.nombre = nombre;
        this.apellido = apellido;
    }

    public LinkedList<Tarea> getTareas() {
        return tareas;
    }

    public double getHsTareasTerminadas() {
        double total = 0;
        for (Tarea t : tareas) {
            if (t.getTerminada()) {
                total += t.getDuracion();
            }
        }
        return total;
    }
}
    public class Tarea {

    private String nombre;
    private String descripcion;
    private double duracionHs;
    private boolean terminada;

    public Tarea(String nombre, String descripcion, double duracionHs, boolean terminada) {
        this.nombre = nombre;
        this.descripcion = descripcion;
        this.duracionHs = duracionHs;
        this.terminada = terminada;
    }

    public double getDuracion() {
        return duracionHs;
    }

    public boolean getTerminada() {
        return terminada;
    }
}
