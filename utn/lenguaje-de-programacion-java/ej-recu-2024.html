    import java.util.ArrayList;

public class Encargado extends Empleado {

    private double porcentajeComision;
    private ArrayList<Evento> eventos = new ArrayList<Evento>();

    public Encargado(String nombre, String apellido, double sueldoHora, double porcentajeComision) {
        super(nombre, apellido, sueldoHora);
        this.porcentajeComision = porcentajeComision;
    }

    public void agregarEvento(Evento evento) {
        eventos.add(evento);
    }

    public double getPagoEncargado() {
        double total = 0;
        for (Evento e : eventos) {
            total += e.getCosto() * porcentajeComision / 100;
        }
        return total;
    }
}

    Evento suma el costo de los menúes y el de los servicios. Le pasa a cada servicio la duración del evento, porque dos de las reglas la usan.

    import java.time.LocalDateTime;
import java.util.ArrayList;

public class Evento {

    private String nombre;
    private double duracionHs;
    private LocalDateTime fechaHora;
    private ArrayList<MenuEvento> menues = new ArrayList<MenuEvento>();
    private ArrayList<Servicio> servicios = new ArrayList<Servicio>();

    public Evento(String nombre, double duracionHs, LocalDateTime fechaHora) {
        this.nombre = nombre;
        this.duracionHs = duracionHs;
        this.fechaHora = fechaHora;
    }

    public void agregarMenu(Menu menu, int cantidad) {
        menues.add(new MenuEvento(menu, cantidad));
    }

    public void agregarServicio(Servicio servicio) {
        servicios.add(servicio);
    }

    public double getCostoMenu() {
        double total = 0;
        for (MenuEvento m : menues) {
            total += m.getCosto();
        }
        return total;
    }

    public double getCostoServicios() {
        double total = 0;
        for (Servicio s : servicios) {
            total += s.getCosto(duracionHs);
        }
        return total;
    }

    public double getCosto() {
        return getCostoMenu() + getCostoServicios();
    }
}
    public class MenuEvento {

    private int cantidad;
    private Menu menu;

    public MenuEvento(Menu menu, int cantidad) {
        this.menu = menu;
        this.cantidad = cantidad;
    }

    public double getCosto() {
        return menu.getValorUnitario() * cantidad;
    }
}
    public class Menu {

        this.descripcion = descripcion;
        this.valorUnitario = valorUnitario;
    }

    public double getValorUnitario() {
        return valorUnitario;

}

    Servicio es abstracta, con getCosto abstracto. Cada tipo de servicio lo redefine con su regla, y Evento los suma sin preguntar de qué tipo es cada uno. Ahí están la herencia, la redefinición y el polimorfismo.

    public abstract class Servicio {

    private String nombre;

    public Servicio(String nombre) {
        this.nombre = nombre;
    }

    public abstract double getCosto(double duracionHs);
}
    import java.util.ArrayList;

public class ServicioPropio extends Servicio {

    private double costoFijo;
    private ArrayList<Empleado> empleados = new ArrayList<Empleado>();

    public ServicioPropio(String nombre, double costoFijo) {
        super(nombre);
        this.costoFijo = costoFijo;
    }

    public void agregarEmpleado(Empleado empleado) {
        empleados.add(empleado);
    }

    @Override
    public double getCosto(double duracionHs) {
        double pagoEmpleados = 0;
        for (Empleado e : empleados) {
            pagoEmpleados += e.getSueldoHora() * duracionHs;
        }
        return costoFijo + pagoEmpleados;
    }
}

    El de terceros no usa la duración, pero la recibe igual para tener la misma firma que los otros.

    public class ServicioDeTerceros extends Servicio {

    private String proveedor;
    private String telefono;
    private double costoBase;

    public ServicioDeTerceros(String nombre, String proveedor, String telefono, double costoBase) {
        super(nombre);
        this.proveedor = proveedor;
        this.telefono = telefono;
        this.costoBase = costoBase;
    }

    @Override
    public double getCosto(double duracionHs) {
        return costoBase;
    }
}
    public class MusicaEnVivo extends Servicio {

    private double costoHora;
    private double gastosFijos;

    public MusicaEnVivo(String nombre, double costoHora, double gastosFijos) {
        super(nombre);
        this.costoHora = costoHora;
        this.gastosFijos = gastosFijos;
    }

    @Override
    public double getCosto(double duracionHs) {
        return duracionHs * costoHora + gastosFijos;
    }
}

    Recorre la LinkedList y se queda con la prueba de menor tiempoSegundos. Si el velocista no tiene pruebas, devuelve null.

    import java.util.LinkedList;

public class Velocista {

    private String nombreCompleto;
    private LinkedList<Prueba> pruebas = new LinkedList<Prueba>();

    public Velocista(String nombreCompleto) {
        this.nombreCompleto = nombreCompleto;
    }

    public void agregarPrueba(Prueba prueba) {
        pruebas.add(prueba);
    }

    public Prueba mejorPrueba() {
        Prueba mejor = null;
        for (Prueba p : pruebas) {
            if (mejor == null || p.getTiempoSegundos() < mejor.getTiempoSegundos()) {
                mejor = p;
            }
        }
        return mejor;
    }
}
    import java.time.LocalDate;

        this.tiempoSegundos = tiempoSegundos;
    }

    public LocalDate getFecha() {
        return fecha;
    }

    public double getTiempoSegundos() {
        return tiempoSegundos;
    }
}
