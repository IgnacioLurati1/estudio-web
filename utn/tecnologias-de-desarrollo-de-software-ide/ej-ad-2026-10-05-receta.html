    El contexto hereda de DbContext, tiene un DbSet por tabla y crea la base con EnsureCreated, como en el laboratorio de la Unidad 4. Va en una carpeta Datos de Perez.Api, junto con el repositorio, que es el único que usa el contexto.

    using Microsoft.EntityFrameworkCore;
using Perez.Dominio;

namespace Perez.Api.Datos
{
    public class RecetaContext : DbContext
    {
        public RecetaContext(DbContextOptions<RecetaContext> options) : base(options)
        {
            this.Database.EnsureCreated();
        }

        public DbSet<Receta> Recetas { get; set; }
    }
}
    using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using Microsoft.EntityFrameworkCore;
using Perez.Dominio;

namespace Perez.Api.Datos
{
    public class RecetaRepositorio : IRecetaRepositorio
    {
        private readonly RecetaContext _context;

        public RecetaRepositorio(RecetaContext context)
        {
            _context = context;
        }

        public async Task<List<Receta>> ObtenerPorEstadoAsync(string estado)
        {
            return await _context.Recetas.Where(r => r.Estado == estado).ToListAsync();
        }

        public async Task<Receta?> ObtenerPorIdAsync(int id)
        {
            return await _context.Recetas.FindAsync(id);
        }

        public async Task AgregarAsync(Receta receta)
        {
            _context.Recetas.Add(receta);
            await _context.SaveChangesAsync();
        }

        public async Task ActualizarAsync(Receta receta)
        {
            _context.Recetas.Update(receta);
            await _context.SaveChangesAsync();
        }
    }
}

    2. API de servicios

    Los endpoints son asíncronos, como pide el enunciado, con la API mínima del laboratorio 1 de la Unidad 6. El repositorio llega por parámetro, porque está registrado en Program.cs.

    using Microsoft.EntityFrameworkCore;
using Perez.Api.Datos;
using Perez.Dominio;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();
builder.Services.AddDbContext<RecetaContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("dbReceta")));
builder.Services.AddScoped<IRecetaRepositorio, RecetaRepositorio>();

var app = builder.Build();

if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

app.UseHttpsRedirection();

app.MapGet("/recetas", async (string estado, IRecetaRepositorio repositorio) =>
    Results.Ok(await repositorio.ObtenerPorEstadoAsync(estado)));

app.MapGet("/recetas/{id}", async (int id, IRecetaRepositorio repositorio) =>
    await repositorio.ObtenerPorIdAsync(id) is Receta receta
        ? Results.Ok(receta)
        : Results.NotFound());

app.MapPost("/recetas", async (Receta receta, IRecetaRepositorio repositorio) =>
{
    List<string> errores = ValidadorReceta.Validar(receta);
    if (errores.Count > 0)
    {
        return Results.BadRequest(errores);
    }
    receta.Id = 0;
    receta.Estado = "Emitida";
    await repositorio.AgregarAsync(receta);
    return Results.Created($"/recetas/{receta.Id}", receta);
});

app.MapPut("/recetas/{id}/dispensar", async (int id, IRecetaRepositorio repositorio) =>
{
    Receta? receta = await repositorio.ObtenerPorIdAsync(id);
    if (receta is null)
    {
        return Results.NotFound();
    }
    if (receta.Estado != "Emitida")
    {
        return Results.BadRequest(new List<string> { "Solo se puede dispensar una receta emitida." });
    }
    receta.Estado = "Dispensada";
    await repositorio.ActualizarAsync(receta);
    return Results.NoContent();
});

app.Run();

    3. Validaciones

    Van en la API, porque según las notas de la cátedra «nuestra API de servicios nunca puede confiarse de que la presentación va a realizar estas validaciones». Un método junta todos los errores, así la pantalla los muestra juntos.

    using System;
using System.Collections.Generic;

namespace Perez.Dominio
{
    public static class ValidadorReceta
    {
        public static List<string> Validar(Receta receta)
        {
            List<string> errores = new List<string>();
            if (string.IsNullOrWhiteSpace(receta.DniPaciente))
            {
                errores.Add("El DNI del paciente es obligatorio.");
            }
            if (string.IsNullOrWhiteSpace(receta.Cobertura))
            {
                errores.Add("La cobertura es obligatoria.");
            }
            if (receta.Cantidad < 1 || receta.Cantidad > 5)
            {
                errores.Add("La cantidad tiene que ser un entero entre 1 y 5.");
            }
            DateTime hoy = DateTime.Today;
            if (receta.FechaEmision.Date != hoy && receta.FechaEmision.Date != hoy.AddDays(-1))
            {
                errores.Add("La fecha de emisión tiene que ser la de hoy o la del día anterior.");
            }
            return errores;
        }
    }
}

    «Igual a la fecha de hoy o el día previo» se compara sin la hora, con .Date. La cobertura no puede quedar vacía, y en la pantalla se elige de una lista con los tres valores de la tabla.

    4. Presentación

    Una clase Conexion con un único HttpClient, con el patrón Singleton como en el laboratorio 3 de la Unidad 6. El puerto 7011 es el del laboratorio. Cambialo por el de applicationUrl en Properties/launchSettings.json de tu API.

    using System;
using System.Net.Http;

namespace Perez.Presentacion
{
    public class Conexion
    {
        private static Conexion? _instancia;
        public HttpClient Cliente { get; }

        private Conexion()
        {
            Cliente = new HttpClient();
            Cliente.BaseAddress = new Uri("https://localhost:7011/");
        }

        public static Conexion Instancia
        {
            get
            {
                if (_instancia == null)
                {
                    _instancia = new Conexion();
                }
                return _instancia;
            }
        }
    }
}

    Otra clase consume los servicios. Los métodos son asíncronos porque esperan a la API.

    using System.Collections.Generic;
using System.Net.Http;
using System.Text;
using System.Threading.Tasks;
using Newtonsoft.Json;
using Perez.Dominio;

namespace Perez.Presentacion
{
    public static class RecetaApi
    {
        public static async Task<List<Receta>> GetPorEstadoAsync(string estado)
        {
            var response = await Conexion.Instancia.Cliente.GetStringAsync($"recetas?estado={estado}");
            return JsonConvert.DeserializeObject<List<Receta>>(response) ?? new List<Receta>();
        }

        public static async Task<List<string>> AgregarAsync(Receta receta)
        {
            var json = JsonConvert.SerializeObject(receta);
            var contenido = new StringContent(json, Encoding.UTF8, "application/json");
            var response = await Conexion.Instancia.Cliente.PostAsync("recetas", contenido);
            if (response.IsSuccessStatusCode)
            {
                return new List<string>();
            }
            var texto = await response.Content.ReadAsStringAsync();
            return JsonConvert.DeserializeObject<List<string>>(texto) ?? new List<string>();
        }

        public static async Task<bool> DispensarAsync(int id)
        {
            var response = await Conexion.Instancia.Cliente.PutAsync($"recetas/{id}/dispensar", null);
            return response.IsSuccessStatusCode;
        }
    }
}

    Los manejadores de eventos son async, que es el único caso donde vale async void.

    using System;
using System.Net.Http;
using System.Threading.Tasks;
using System.Windows.Forms;
using Perez.Dominio;

namespace Perez.Presentacion
{
    public partial class FormRecetas : Form
    {
        public FormRecetas()
        {
            InitializeComponent();
            cmbEstado.DropDownStyle = ComboBoxStyle.DropDownList;
            cmbEstado.Items.Add("Emitida");
            cmbEstado.Items.Add("Dispensada");
            cmbEstado.SelectedIndex = 0;
        }

        private async void FormRecetas_Load(object? sender, EventArgs e)
        {
            await CargarAsync();
        }

        private async void btnBuscar_Click(object? sender, EventArgs e)
        {
            await CargarAsync();
        }

        private async Task CargarAsync()
        {
            try
            {
                dgvRecetas.DataSource = await RecetaApi.GetPorEstadoAsync(cmbEstado.Text);
            }
            catch (HttpRequestException)
            {
                MessageBox.Show("No se pudo conectar con la API. Fijate que esté corriendo.");
            }
        }

        private async void btnAgregar_Click(object? sender, EventArgs e)
        {
            FormAltaReceta alta = new FormAltaReceta();
            if (alta.ShowDialog() == DialogResult.OK)
            {
                await CargarAsync();
            }
        }

        private async void btnDispensar_Click(object? sender, EventArgs e)
        {
            if (dgvRecetas.CurrentRow?.DataBoundItem is not Receta receta)
            {
                MessageBox.Show("Elegí una receta del listado.");
                return;
            }
            if (receta.Estado != "Emitida")
            {
                MessageBox.Show("Solo se puede dispensar una receta emitida.");
                return;
            }
            if (await RecetaApi.DispensarAsync(receta.Id))
            {
                await CargarAsync();
            }
            else
            {
                MessageBox.Show("No se pudo dispensar la receta.");
            }
        }
    }
}

    using System;
using System.Collections.Generic;
using System.Net.Http;
using System.Windows.Forms;
using Perez.Dominio;

namespace Perez.Presentacion
{
    public partial class FormAltaReceta : Form
    {
        public FormAltaReceta()
        {
            InitializeComponent();
            cmbCobertura.DropDownStyle = ComboBoxStyle.DropDownList;
            cmbCobertura.Items.AddRange(new object[] { "Obra Social", "Prepaga", "Particular" });
            nudCantidad.Minimum = 1;
            nudCantidad.Maximum = 5;
            dtpFechaEmision.Value = DateTime.Today;
        }

        private async void btnGuardar_Click(object? sender, EventArgs e)
        {
            Receta receta = new Receta
            {
                DniPaciente = txtDniPaciente.Text.Trim(),
                FechaEmision = dtpFechaEmision.Value.Date,
                Cantidad = (int)nudCantidad.Value,
                Cobertura = cmbCobertura.Text

            try
            {
                List<string> errores = await RecetaApi.AgregarAsync(receta);
                if (errores.Count == 0)
                {
                    DialogResult = DialogResult.OK;
                }
                else
                {
                    MessageBox.Show(string.Join(Environment.NewLine, errores), "No se pudo guardar");
                }
            }
            catch (HttpRequestException)
            {
                MessageBox.Show("No se pudo conectar con la API. Fijate que esté corriendo.");
            }
        }
    }
}
